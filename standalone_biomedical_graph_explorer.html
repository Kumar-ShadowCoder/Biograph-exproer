backend/main.py (FastAPI + Official Neo4j Driver) Python 3.11+ 
from fastapi import FastAPI, Query
from fastapi.middleware.cors import CORSMiddleware
from neo4j import GraphDatabase
import os

app = FastAPI(title="Biomedical Knowledge Graph API", version="1.0.0")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

driver = GraphDatabase.driver(
    os.getenv("NEO4J_URI", "bolt://localhost:7687"),
    auth=(os.getenv("NEO4J_USER", "neo4j"), os.getenv("NEO4J_PASS", "password"))
)

@app.get("/api/graph/{entity_name}")
def get_neighborhood(entity_name: str, limit: int = 50):
    query = "MATCH (source {name: $name})-[r]-(target) RETURN source, r, target LIMIT $limit"
    with driver.session() as session:
        result = session.run(query, name=entity_name, limit=limit)
        nodes, links = {}, []
        for record in result:
            for key in ["source", "target"]:
                n = record[key]
                nodes[n["name"]] = {"id": n["name"], "type": list(n.labels)[0], **dict(n)}
            rel = record["r"]
            links.append({"source": rel.start_node["name"], "target": rel.end_node["name"], "label": rel.type})
        return {"nodes": list(nodes.values()), "links": links}
database/repositioning.cypher (Neo4j 5.x) Cypher Query 
// Multi-Hop Drug Repositioning Query (2-Hop Shared Gene Targets)
MATCH (c:Drug)-[:TARGETS]->(g:Gene)-[:ASSOCIATED_WITH]->(d:Disease)
WHERE NOT (c)-[:TREATS]->(d)
WITH c, d, collect(g.name) AS shared_genes, count(g) AS gene_overlap
WHERE gene_overlap >= 1
RETURN c.name AS candidate_drug, d.name AS target_disease, shared_genes, gene_overlap
ORDER BY gene_overlap DESC LIMIT 25;
